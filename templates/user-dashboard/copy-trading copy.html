Are you sure you want to stop copying ${traderName}?

Your current positions will remain open. You can close them manually.

`, icon: 'warning', showCancelButton: true, confirmButtonText: 'Yes, Stop Copying', cancelButtonText: 'Cancel', confirmButtonColor: '#ef4444', cancelButtonColor: '#6b7280', }); if (!confirmed.isConfirmed) { return; } this.showLoading('Unfollowing trader...'); // Unfollow trader await this.copyAPI.unfollowTrader(subscriptionId); // Update local state this.subscriptions = this.subscriptions.filter(s => s.id !== subscriptionId); this.updateDashboard(); // Show success message this.showSuccess('Successfully stopped copying trader'); // Refresh page setTimeout(() => { window.location.reload(); }, 1500); } catch (error) { console.error('Failed to unfollow trader:', error); this.showError(error.message || 'Failed to unfollow trader'); } finally { this.hideLoading(); } } async toggleSubscription(subscriptionId) { try { this.showLoading('Updating subscription...'); const response = await this.copyAPI.toggleSubscription(subscriptionId); // Update local state const subscription = this.subscriptions.find(s => s.id === subscriptionId); if (subscription) { subscription.is_active = response.is_active; } this.updateDashboard(); const message = response.is_active ? } finally { this.hideLoading(); } } async updateSubscriptionSettings(subscriptionId, settings) { try { // Validate settings const validation = this.copyAPI.validateSubscriptionSettings(settings); if (!validation.valid) { throw new Error(validation.errors.join(', ')); } this.showLoading('Updating settings...'); const updated = await this.copyAPI.updateSubscription(subscriptionId, settings); // Update local state const index = this.subscriptions.findIndex(s => s.id === subscriptionId); if (index !== -1) { this.subscriptions[index] = updated; } this.updateDashboard(); this.showSuccess('Settings updated successfully'); return updated; } catch (error) { console.error('Failed to update settings:', error); this.showSuccess('Trade rejected'); // Reload pending trades await this.loadPendingTrades(); this.updatePendingTradesUI(); } catch (error) { console.error('Failed to reject trade:', error); this.showError(error.message || 'Failed to reject trade');
